# Tennis Architecture — Sources/Tennis [Tennis]
# Senior Engineer design doc. Format mirrors .map files.
# >> lines show key outgoing dependencies and callers.
# This is the durable runtime replacing the M1 TennisScene shell.
#
# ─────────────────────────────────────────────────────────────────────────────
# DESIGN PRINCIPLES
# ─────────────────────────────────────────────────────────────────────────────
#   Data-driven: shot + surface params in tunable tables. Swap to JSON5 = one change.
#   No floats in simulation: all positions and velocities are Int * 100 fixed-point.
#   Deterministic: fixed-tick only, no Float-branching, no SDL_GetTicks64 in logic.
#   Mirror SpaceInvaders: TennisApp → EngineWrapped → TennisScene.
#   Minimal abstraction: no IBall, no IController protocol, no PhysicsWorld.
#     Exception: HumanController vs OpponentAI ARE different concrete types.
#     They don't share a protocol — TennisScene knows about both explicitly.
#     Reason: adding a second human player or network controller is a future
#     concern. Protocol now = abstraction with no current beneficiary.
#   Operations belong on the types that own the data.
#     Ball owns its own physics step, hit application, collision checks.
#     TennisPlayer owns movement, hitstop ticking, hit resolution.
#     No separate "Sim" or "System" objects that duplicate the data's natural home.
#
# PLAYER / CONTROLLER SPLIT (key architectural decision)
#   TennisPlayer = state + operations on that state (move, tick, tryHit)
#     - no input methods, no scene ref, no AnyObject constraint
#   HumanController = class, reads SDL input, drives its TennisPlayer
#     - implements IVirutalControllerListener (requires AnyObject/weak ref)
#   OpponentAI = struct, reads ball + scene state, drives its TennisPlayer
#     - no input path; symmetric shot vocabulary with player
#   TennisScene owns:
#     playerState: TennisPlayer     // human player state + behavior
#     opponentState: TennisPlayer   // CPU player state + behavior
#     humanController: HumanController
#     ai: OpponentAI
#   Both players use identical tryHit() on TennisPlayer — symmetry enforced structurally.
#
# SERVING MECHANICS (full spec)
#   Phase = .serving(side)
#   Server locked at CourtLayout.servePosition(side). Movement disabled.
#   Dpad before action press selects serve direction (aim within service box).
#   Human: press A → HumanController sets playerState.pendingShot = .serve + target.
#   CPU: OpponentAI.serveTick() → sets opponentState.pendingShot = .serve + center target.
#   TennisScene.logic() sees pendingShot == .serve:
#     → ball.applyHit(shots[.serve], ...) → ball launched
#     → phase = .serveInFlight(serverSide)
#   Phase = .serveInFlight(from: Side)
#     ball.step(surface) runs normally.
#     On first bounce (z hits 0 in step):
#       ball.isServeLegal(serverSide) → true: phase = .rally, ball.isServe = false
#                                     → false: phase = .pointEnd(receiver, .serveIllegal)
#     ball.checkNetCollision() → .pointEnd(receiver, .netFault)
#   Phase = .rally: normal hit/bounce/fault logic.
#
# WIRING DIAGRAM
#   TennisApp
#     mounts ExternalFiles/ via VirtualDrive
#     creates FullWindow(160×144)
#     creates MatchConfig (surface selection)
#     creates TennisScene(config)
#     creates EngineWrapped(scene)
#     addFixedListener(engineWrapped, msPerTick:16)   ← fixed-tick simulation
#     addEventListener(scene)                          ← raw SDL events
#     window.drawable = engineWrapped                 ← draw path
#     window.displayRenderClient.addResourceCache(scene)
#
#   Per tick (62.5fps fixed tick at 16ms):
#     Application → EngineWrapped.step(delta) → TennisScene.step(delta) → input dispatch
#                                              → TennisScene.logic(delta) → sim step
#     Application → EngineWrapped.draw(delta) → TennisScene.draw(delta, renderer)
#     Application → TennisScene.onEvents(events) → HumanController input feed

Side
  top                   // CPU side (upper half of court; y increases downward)
  bottom                // human player (lower half)
  var opposite: Side

ShotType                // 6 rally shots + serve
  topspin               // A — fast, high arc, high bounce
  slice                 // B — low arc, low skid bounce
  flat                  // A+B outside smash context
  lob                   // A → B — high arc, long hang, deep bias
  dropShot              // B → A — short travel, very low second bounce
  smash                 // A+B in overhead context (z > smashHeightThreshold)
  serve                 // fixed baseline shot; own ShotDef entry; different arc

SpinType                // stored on Ball; affects post-bounce behavior
  topspin               // high bounce, maintain speed
  slice                 // low bounce, skid (reduced postBounceSpeedMult)
  none                  // flat / lob / smash / serve

ShotQuality
  perfect               // within hitRadiusPerfect: full speed + placement
  good                  // within hitRadiusGood: standard
  late                  // within hitRadiusLate: weakened

MatchPhase
  serving(side: Side)
  serveInFlight(from: Side)
  rally
  pointEnd(winner: Side, reason: PointEndReason)
  matchEnd(winner: Side)

PointEndReason
  doubleBounce
  netFault
  outOfBounds
  serveIllegal

PlayerAnim                  // player sprite animation state machine
  idle
  walkN; walkS; walkE; walkW; walkNE; walkNW; walkSE; walkSW
  swing                     // hit animation; driven by hitstop window
  serve                     // serve wind-up pose

  ticksPerFrame() -> Int
    // idle=12, walk=6, swing=3, serve=8
    // faster anims = more responsive feel at 160×144

PlayerStats
  power: Int        // 1–10: shot speed multiplier
  speed: Int        // 1–10: movement pixels per tick
  control: Int      // 1–10: aim cone tightness
  spin: Int         // 1–10: topspin bounce amplifier / slice skid reduction

  static balanced: PlayerStats    // 6/6/6/6
  static power: PlayerStats       // 9/4/5/7

  moveSpeed() -> Int
    // pixels per tick this player moves: 1 + (speed * 2) / 10
    // range: speed=1 → 1px/tick, speed=10 → 3px/tick

  shotSpeedMult() -> Int
    // fixed-point multiplier * 100: 100 + (power - 5) * 8
    // range: power=1 → 68, power=10 → 140

  spinBounceMult() -> Int
    // * 100: 100 + (spin - 5) * 6  (applied to bounceHeight on topspin)

  aimSpread() -> Int
    // pixels of random spread on Late quality: (10 - control) * 3

# ─────────────────────────────────────────────────────────────────────────────
# Sources/Tennis/Game/CourtLayout.swift
# ─────────────────────────────────────────────────────────────────────────────
# All geometry for 160×144. Caseless enum = zero instances, pure namespace.
# Nothing else hardcodes court geometry. Single source of truth.

enum CourtLayout
  // ── Rects (screen pixels; NOT fixed-point)
  viewport: Rect<Int>               // (0, 0, 160, 144)
  court: Rect<Int>                  // (24, 10, 112, 124)
  serviceBoxTop: Rect<Int>
  serviceBoxBottom: Rect<Int>
  singlesBounds: Rect<Int>

  // ── Key coordinates
  netY: Int                         // 72
  netThickness: Int                 // 2
  baselineTop: Int                  // court.y + 2
  baselineBottom: Int               // court.bottom - 2
  sidelineLeft: Int                 // court.x + 2
  sidelineRight: Int                // court.right - 2

  // ── Serve
  servePositionBottom: Point<Int>   // (80, 130)
  servePositionTop: Point<Int>      // (80, 14)
  static func servePosition(side: Side) -> Point<Int>
  static func serviceBox(receiverSide: Side) -> Rect<Int>

  // ── Hit radii (screen pixels)
  hitRadiusPerfect: Int             // 3
  hitRadiusGood: Int                // 7
  hitRadiusLate: Int                // 11  — beyond this: no hit

  // ── Smash
  smashHeightThreshold: Int         // z * 100 threshold; A+B above this → .smash

  // ── Timing
  chargeFramesMax: Int              // 30 ticks ≈ 480ms
  hitstopFrames: Int                // 6 ticks; both players freeze on any hit
  pointEndPauseTicks: Int           // 45 ticks ≈ 720ms before next serve

  // ── Physics
  gravity: Int                      // vz decremented by this each tick * 100; e.g. 80

# ─────────────────────────────────────────────────────────────────────────────
# Sources/Tennis/Game/Ball.swift
# ─────────────────────────────────────────────────────────────────────────────
# Value type. Owns its physics, collision checks, and render coordinate helpers.
# Sub-pixel coords (Int * 100) → screen: divide by 100.
# Screen render Y: screenY - z/100 (height lifts sprite upward).

Ball
  // ── simulation state (fixed-point * 100)
  x: Int              // court x * 100
  y: Int              // court y * 100 (increases downward)
  z: Int              // height above court * 100 (0 = ground)
  vx: Int             // x velocity * 100 per tick
  vy: Int             // y velocity * 100 per tick (positive = toward bottom)
  vz: Int             // vertical velocity * 100 per tick (positive = rising)
  currentBounceVz: Int  // vz to restore on each bounce; set by applyHit, reused per bounce
  currentPostBounceSpeedMult: Int  // speed fraction after bounce; set by applyHit

  // ── logical state
  ownerSide: Side     // last side to hit; determines legality
  spin: SpinType
  bounceCount: Int    // on current owner's side; 2 = double bounce = point over
  isServe: Bool       // true from serve launch until first legal bounce

  // ── derived / render helpers
  landingX: Int       // projected bounce x * 100; updated each tick by step()
  landingY: Int       // projected bounce y * 100
  trailColor: SDLColor
  trailHistory: [(x: Int, y: Int)]  // last 3 screen positions (already /100); push on bounce
  rotation: Int           // visual-only; degrees * 100; updated in step(); not simulation state
                          // feeds DrawCmd.rotation at draw time as Float(rotation)/100.0

  // ── computed render positions (screen pixels, NOT fixed-point)
  var screenPos: Point<Int>         // Point(x/100, y/100 - z/100)
  var shadowPos: Point<Int>         // Point(x/100, y/100)   — shadow stays on ground
  var shadowAlpha: Float            // 1.0 - (z/100) / 50.0, clamped 0.2..1.0

  static var atRest: Ball
    // Ball at court center; all velocities 0; ownerSide=.bottom

  mutating func step(surface: SurfaceDef)
    // 1. x += vx; y += vy; z += vz
    // 2. vz -= CourtLayout.gravity
    // 3. on z <= 0 (bounce):
    //      z = 0
    //      vz = currentBounceVz
    //      vx = (vx * currentPostBounceSpeedMult * surface.speedMult) / 10000
    //      vy = (vy * currentPostBounceSpeedMult * surface.speedMult) / 10000
    //      bounceCount++
    //      prepend screenPos to trailHistory (keep last 3)
    // 4. landingX, landingY = projectLanding()
    // 5. rotation += (abs(vx) + abs(vy)) / 200   (visual spin; speed drives rotation rate)
    //    rotation %= 36000  (wrap at 360°)
    >> CourtLayout.gravity SurfaceDef.getter:speedMult SurfaceDef.getter:bounceMult

  mutating func applyHit(shot: ShotDef, quality: ShotQuality, chargeLevel: Int,
                          targetX: Int, targetY: Int, fromSide: Side, stats: PlayerStats)
    // direction: (targetX - screenPos.x, targetY - screenPos.y), normalized to speed
    // speed = shot.baseSpeed
    //   * qualityMult(quality)           // perfect=100, good=85, late=65  (* /100)
    //   * (100 + chargeLevel * 20 / chargeFramesMax) / 100
    //   * stats.shotSpeedMult() / 100
    // vz = shot.arcHeight
    // currentBounceVz = shot.bounceHeight * stats.spinBounceMult() / 100
    //   (spin stat only amplifies topspin and attenuates slice; ignored for .none)
    // currentPostBounceSpeedMult = shot.postBounceSpeedMult
    // spin = shot.spinType
    // trailColor = shot.trailColor()
    // ownerSide = fromSide
    // bounceCount = 0
    // isServe = (shot.shotType == .serve)
    // prepend current screenPos to trailHistory
    >> ShotDef.trailColor() PlayerStats.shotSpeedMult() PlayerStats.spinBounceMult()
    >> CourtLayout.chargeFramesMax

  func projectLanding() -> Point<Int>
    // ticksToLand = vz / CourtLayout.gravity  (integer; approximate fine here)
    // returns Point((x + vx * ticksToLand) / 100, (y + vy * ticksToLand) / 100)

  func checkNetCollision() -> Bool
    // shadowPos.y crosses CourtLayout.netY AND z / 100 < CourtLayout.netThickness
    // check: previous y/100 <= netY < current y/100 (or vice versa) while z low

  func isInBounds() -> Bool
    // landing point (landingX/100, landingY/100) within CourtLayout.singlesBounds
    // generous on lines: contains() with no shrink

  func isServeLegal(serverSide: Side) -> Bool
    // landing point within CourtLayout.serviceBox(receiverSide: serverSide.opposite)

  func smashEligible(playerScreenPos: Point<Int>) -> Bool
    // z > CourtLayout.smashHeightThreshold
    // AND dist(playerScreenPos, projectLanding()) <= CourtLayout.hitRadiusGood * 2

TennisPlayer
  pos: Point<Int>          // screen pixels
  side: Side
  stats: PlayerStats
  chargeFrames: Int        // ticks holding shot button; set by controller each tick
  pendingShot: ShotType?   // committed shot type; cleared when tryHit succeeds

  pendingTargetY: Int

  mutating func tickHitstop()
    // if isInHitstop: hitstopTimer--; if hitstopTimer <= 0: isInHitstop = false
    // animation: animState.tick(.swing) while hitstop active

  mutating func applyMovement(dir: Point<Int>)
    // if isInHitstop: return (locked during hit freeze)
    // pos.x += dir.x * stats.moveSpeed()
    // pos.y += dir.y * stats.moveSpeed()
    // clamp pos to CourtLayout.court (player stays in court)
    // animation: derive PlayerAnim from dir (zero→.idle, else walk direction)
    //   animState.tick(derivedAnim)
    >> PlayerStats.moveSpeed() CourtLayout.court PlayerAnimState.tick(_:)

  func tryHit(ball: Ball) -> HitResult?
    // GUARD: pendingShot != nil, !isInHitstop, ball.ownerSide != side (or isServe)
    // dist = distance(pos, ball.screenPos)
    // GUARD: dist <= CourtLayout.hitRadiusLate
    // quality: dist <= hitRadiusPerfect → .perfect; <= hitRadiusGood → .good; else .late
    // shotType = pendingShot!
    // if shotType == .flat AND ball.smashEligible(pos) → shotType = .smash
    // return HitResult(quality, shotType, pendingTargetX, pendingTargetY, chargeFrames)
    // NOTE: does NOT clear pendingShot or set isInHitstop — TennisScene does that
    //   after calling ball.applyHit(). Reason: keeps tryHit a pure query.
    >> Ball.smashEligible(_:) CourtLayout (radius constants)

# ─────────────────────────────────────────────────────────────────────────────
# Sources/Tennis/Game/HumanController.swift
# ─────────────────────────────────────────────────────────────────────────────
# Class — CommandRepeater requires AnyObject (weak ref).
# Reads input → drives playerState via TennisPlayer mutating methods.

HumanController < IVirutalControllerListener
  inputBuffer: ShotInputBuffer
  clientId: UInt32
  deviceId: UInt32
  priv moveDir: Point<Int>       // dpad direction this tick; reset each tick

  init(clientId: UInt32, deviceId: UInt32)

  register(with repeater: CommandRepeater)
    >> CommandRepeater.addListener(_:_:_:)

  deregister(from repeater: CommandRepeater)
    >> CommandRepeater.removeListener(_:_:_:)

  tick(player: inout TennisPlayer, phase: MatchPhase,
       opponentPos: Point<Int>, tick: UInt64)
    // 1. player.tickHitstop()
    // 2. if phase == .serving(.bottom):
    //      movement locked
    //      if pendingDecision set in onInput: apply to player.pendingShot + target
    //      serve target: autoAimTarget(player.pos, opponentPos, .serve) offset by dpad
    // 3. else: player.applyMovement(moveDir)
    //      if shotDecision set in onInput:
    //        player.pendingShot = shotDecision
    //        player.pendingTargetX/Y = autoAimTarget(player.pos, opponentPos, shotDecision)
    //          overridden if moveDir != .zero at time of shot press
    //        player.chargeFrames = inputBuffer.chargeLevel(tick)
    >> TennisPlayer.tickHitstop() TennisPlayer.applyMovement(_:)
    >> ShotInputBuffer.chargeLevel(_:)

  onInput(_ controller: VirtualController)
    // called by CommandRepeater; stores decisions for tick() to apply
    // 1. read dpad → set moveDir
    // 2. A+B simultaneously → store .flat decision
    // 3. A or B single press → inputBuffer.press(button, tick) → ShotType?
    //    store resolved ShotType as pending decision
    // 4. button release → inputBuffer.release(button)
    >> ShotInputBuffer.press(_:_:) ShotInputBuffer.release(_:)
    >> VirtualController.getter:state PressedControllerButtons

  priv autoAimTarget(playerPos: Point<Int>, opponentPos: Point<Int>,
                      shotType: ShotType) -> Point<Int>
    // default: corner furthest from opponentPos
    // dpad held at shot press: use moveDir to pick which corner/direction
    // control stat: high = precise corner; low = ± aimSpread() px variation
    // lob always targets near opponent's baseline
    // dropShot always targets near net on opponent's side
    >> PlayerStats.aimSpread() CourtLayout (baseline + net positions)

# ─────────────────────────────────────────────────────────────────────────────
# Sources/Tennis/Game/OpponentAI.swift
# ─────────────────────────────────────────────────────────────────────────────
# Drives opponentState. Value type — TennisScene owns one.
# Hidden information: uses ball.projectLanding() freely (perfect prediction — OK per qa).
# No physics cheat: movement uses stats.moveSpeed() per tick, no teleport.
# Same ShotDef table as player — symmetric.

OpponentAI
  targetPos: Point<Int>       // where AI is moving toward this rally
  pendingDecision: ShotType?  // chosen next shot; committed early, applied on hit

  mutating func tick(opponent: inout TennisPlayer, ball: Ball, playerPos: Point<Int>,
                     phase: MatchPhase, rallyCount: Int, tick: UInt64)
    // 1. opponent.tickHitstop()
    // 2. if phase == .serving(.top): serveTick(&opponent)
    // 3. else:
    //      if ball.ownerSide == .bottom: targetPos = ball.projectLanding()
    //      opponent.applyMovement(directionTo(targetPos))
    //      if in hit range AND pendingDecision == nil:
    //        pendingDecision = selectShot(ball, opponent.pos, playerPos, rallyCount, tick)
    //      set opponent.pendingShot = pendingDecision + target (center of player court)
    >> TennisPlayer.tickHitstop() TennisPlayer.applyMovement(_:)
    >> Ball.projectLanding() CourtLayout (movement bounds)

  priv mutating func serveTick(_ opponent: inout TennisPlayer)
    // opponent.pendingShot = .serve
    // target = center of human player's service box
    >> CourtLayout.serviceBox(receiverSide:)

  priv func selectShot(ball: Ball, aiPos: Point<Int>, playerPos: Point<Int>,
                        rallyCount: Int, tick: UInt64) -> ShotType
    // DEFAULT: .topspin
    // playerPos.y < CourtLayout.netY + 10 AND rallyCount > 3 → .lob
    // playerPos.y > CourtLayout.baselineBottom - 10 AND rallyCount > 4
    //   AND tick % 5 == 0 → .dropShot   (deterministic stand-in for probability)
    // ball.smashEligible(aiPos) → .smash
    >> Ball.smashEligible(_:) CourtLayout.netY CourtLayout.baselineBottom

# - Fixed-tick only: game logic NEVER calls SDL_GetTicks64() or reads wall time
# - No floats in simulation: Int * 100 fixed-point; no Float comparisons in logic
# - No Process/subprocess: pure Swift/Foundation only
# - VirtualDrive for all assets: no bare FileManager in game code
# - Ball legality judged at landing point (ball.landingX/Y), not ball center
# - Hit check is radius + quality penalty — never a hard-fail gate
# - AI uses identical ShotDef table as player — no asymmetric physics
# - AI may call ball.projectLanding() freely — hidden info advantage is OK
# - AI may NOT teleport — movement always via TennisPlayer.applyMovement()
# - Serve always from fixed position — no foot fault, no second serve
